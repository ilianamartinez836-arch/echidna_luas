local NAME = "autocut"

local PT_GUARD_FULL = "echidna_autocut_pt_guard_full.txt"

local CFG = { enabled = true, switch = true, switch_back = false, wall = true, debug = false }

local LEAD, TILT, PRESWAP, BACKOFF, AIM_TOL, GRACE = 0.15, 25, 0.1, 3, 10, 0.35

local sqrt, abs, max, min, floor = math.sqrt, math.abs, math.max, math.min, math.floor
local atan2 = math.atan2 or math.atan
local deg, rad = math.deg, math.rad

local PL = "CTerrorPlayer"
local IN_ATTACK = IN_ATTACK or 1
local GROUP_SV = GROUP_PLAYER_SURVIVOR or 1
local ZC_SMOKER = 1
local MELEE_CLASS = "CTerrorMeleeWeapon"

local function log(s) pcall(client.print, "[" .. NAME .. "] " .. tostring(s)) end
local function dbg(s) if CFG.debug then log(s) end end
local function V(x, y, z) return { x = x, y = y, z = z } end
local function dist(a, b) local dx, dy, dz = a.x - b.x, a.y - b.y, a.z - b.z; return sqrt(dx * dx + dy * dy + dz * dz) end
local function clamp(v, lo, hi) if v < lo then return lo elseif v > hi then return hi end return v end
local function has_bit(v, b) return floor(v / b) % 2 == 1 end
local function angles_to(from, to)
    local dx, dy, dz = to.x - from.x, to.y - from.y, to.z - from.z
    return deg(atan2(-dz, sqrt(dx * dx + dy * dy))), deg(atan2(dy, dx))
end
local function median(t)
    local c = {}
    for i = 1, #t do c[i] = t[i] end
    table.sort(c)
    local n = #c
    if n == 0 then return nil end
    return (n % 2 == 1) and c[(n + 1) / 2] or (c[n / 2] + c[n / 2 + 1]) / 2
end

local maxc = 32
local function handle_index(h)
    if type(h) ~= "number" or h < 0 then return nil end
    local i = h % 2048
    if i < 1 or i >= 2047 then return nil end
    return i
end
local function player_handle(h)
    local i = handle_index(h)
    if i and i <= maxc then return i end
    return nil
end

local function cvar(name, fallback)
    local v
    pcall(function() v = Cvars.FindVar(name):GetFloat() end)
    if type(v) ~= "number" or v <= 0 then return fallback end
    return v
end

local IPT = 1 / 30
pcall(function() local v = GlobalVars.interval_per_tick(); if type(v) == "number" and v > 0 then IPT = v end end)

local function server_now(lp)
    local tb
    pcall(function() tb = lp:get_prop_int(PL, "m_nTickBase") end)
    if type(tb) ~= "number" then return nil end
    return tb * IPT
end

local set_style = nil
local function set_angles(cmd, ang)
    if set_style ~= 2 and pcall(function() cmd:set_viewangles(ang) end) then set_style = 1; return true end
    if set_style ~= 1 and pcall(function() cmd:set_viewangles(ang.x, ang.y, ang.z) end) then set_style = 2; return true end
    return false
end

local function keep_movement(cmd, old_yaw, new_yaw)
    pcall(function()
        local fm, sm = cmd:get_forwardmove(), cmd:get_sidemove()
        local r = rad(new_yaw - old_yaw)
        local c, s = math.cos(r), math.sin(r)
        cmd:set_forwardmove(fm * c - sm * s)
        cmd:set_sidemove(fm * s + sm * c)
    end)
end

local function all_infected()
    local out, have = {}, {}
    local function add(list, need_team)
        if type(list) ~= "table" then return end
        for _, p in ipairs(list) do
            local ok, idx = pcall(function() return p:entindex() end)
            if ok and idx and not have[idx] then
                local keep = true
                if need_team then
                    local okt, tm = pcall(function() return p:get_team() end)
                    keep = okt and tm == 3
                end
                if keep then have[idx] = true; out[#out + 1] = p end
            end
        end
    end
    pcall(function() add(EntityCache.GetGroup(GROUP_SPECIAL_INFECTED or 4), false) end)
    pcall(function() add(EntityCache.GetGroup(GROUP_PLAYER_INFECTED or 2), true) end)
    pcall(function() add(EntityCache.GetGroup(GROUP_PLAYER_ALL or 0), true) end)
    return out
end

local function save_cfg()
    pcall(function()
        local f = io.open(CFG_FILE, "w")
        for k, v in pairs(CFG) do f:write(k, "=", tostring(v), "\n") end
        f:close()
    end)
end
pcall(function()
    local f = io.open(CFG_FILE, "r")
    if not f then return end
    for line in f:lines() do
        local k, v = line:match("^([%w_]+)=(.+)$")
        if k and CFG[k] ~= nil then CFG[k] = (v == "true") end
    end
    f:close()
end)

local function read_file(file)
    local c = ""
    pcall(function() local f = io.open(file, "r"); if f then c = f:read("*a") or ""; f:close() end end)
    return c
end
local function write_file(file, txt)
    pcall(function() local f = io.open(file, "w"); if f then f:write(txt); f:close() end end)
end
local function crashed_last_time(file)
    local c = read_file(file)
    return c:find("TRY", 1, true) and not c:find("OK", 1, true)
end

local PT_MASK = 0x1000B
local pt = { ok = nil, calls = 0, crashed = crashed_last_time(PT_GUARD),
             full_ok = nil, full_calls = 0, full_crashed = crashed_last_time(PT_GUARD_FULL) }
local function bits_solid(r)
    r = floor(r)
    for _, b in ipairs({ 0x1, 0x2, 0x8, 0x10000 }) do
        if has_bit(r, b) then return true end
    end
    return false
end
local function ask(name, x, y, z)
    return pcall(function() return EngineTrace[name](V(x, y, z), PT_MASK) end)
end
local function solid_at(x, y, z)
    if pt.full_ok ~= false and not pt.full_crashed then
        local first = pt.full_calls == 0
        if first then write_file(PT_GUARD_FULL, "TRY\n") end
        local ok, r = ask("GetPointContents", x, y, z)
        if first then write_file(PT_GUARD_FULL, "TRY\nOK\n") end
        pt.full_calls = pt.full_calls + 1
        if ok and type(r) == "number" then pt.full_ok = true; return bits_solid(r) end
        if pt.full_calls >= 3 and not pt.full_ok then pt.full_ok = false end
    end
    if pt.crashed or pt.ok == false then return nil end
    local first = pt.calls == 0
    if first then write_file(PT_GUARD, "TRY\n") end
    local ok, r = ask("GetPointContents_WorldOnly", x, y, z)
    if first then write_file(PT_GUARD, "TRY\nOK\n") end
    pt.calls = pt.calls + 1
    if not ok or type(r) ~= "number" then
        if pt.calls >= 3 and not pt.ok then pt.ok = false end
        return nil
    end
    pt.ok = true
    return bits_solid(r)
end

local srv = { next = 0, speed = 1000, range = 750 }
local function refresh(now)
    if now < srv.next then return end
    srv.next = now + 5
    srv.speed = cvar("tongue_fly_speed", 1000)
    srv.range = cvar("tongue_range", 750)
    pcall(function()
        local m = EngineClient.GetMaxClients()
        if type(m) == "number" and m > 0 and m < 64 then maxc = m end
    end)
end

local corr = {}
local function save_calib()
    pcall(function()
        local f = io.open(CALIB_FILE, "w")
        for _, c in ipairs(corr) do f:write(string.format("%.4f\n", c)) end
        f:close()
    end)
end
pcall(function()
    local f = io.open(CALIB_FILE, "r")
    if not f then return end
    for line in f:lines() do local v = tonumber(line); if v then corr[#corr + 1] = v end end
    f:close()
end)
local function correction() return median(corr) or 0 end

local fire_pending = false
local tongue = nil
local stats = { cuts = 0, dragged = 0, last = "-" }
local last_owner = nil
local swapped_by_us, switch_back_at, post_cut_release = false, nil, false
local was_melee, melee_since, cur_cls = false, nil, "unknown"
local override_until, smoker_block, smoker_miss = 0, {}, {}
local preswap_seen, aim_since = -99, nil
local last_swap_cmd = -99
local preswap_block_until, ignore_back_until = 0, 0
local wall_cache = {}

local function weapon_class(lp)
    local cls
    pcall(function()
        local idx = handle_index(lp:get_prop_int(PL, "m_hActiveWeapon"))
        if idx then cls = ClientEntityList.GetClientEntity(idx):get_class_name() end
    end)
    return cls
end
local function melee_out(lp)
    local cls = weapon_class(lp)
    if cls == nil then return true, "unknown" end
    return cls == MELEE_CLASS, cls
end

local function swap_to_melee(now)
    if now - last_swap_cmd < 0.4 then return end
    last_swap_cmd = now
    pcall(function() EngineClient.ClientCmd("use weapon_melee") end)
    swapped_by_us = true
end
local function swap_back(now)
    last_swap_cmd = now
    ignore_back_until = now + 1.0
    pcall(function() EngineClient.ClientCmd("lastinv") end)
end

local function track_weapon(lp, now)
    local _, cls = melee_out(lp)
    cur_cls = cls
    local is_melee = (cls == MELEE_CLASS)
    if cls ~= "unknown" then
        if is_melee and not was_melee then melee_since = now end
        if swapped_by_us and was_melee and not is_melee and melee_since and now - melee_since > 0.15 then
            override_until = now + 2.0
            if tongue then tongue.override = true end
            swapped_by_us = false; switch_back_at = nil
        elseif is_melee and tongue and tongue.override and now >= override_until then
            tongue.override = false
        end
        if was_melee and not is_melee and now > ignore_back_until then
            preswap_block_until = max(preswap_block_until, now + 2.0)
        end
        if not is_melee then melee_since = nil end
        was_melee = is_melee
    end
    return is_melee
end

local function aim_error(s_eye, s_ang, target)
    local p, y = rad(s_ang.x), rad(s_ang.y)
    local fx, fy, fz = math.cos(p) * math.cos(y), math.cos(p) * math.sin(y), -math.sin(p)
    local dx, dy, dz = target.x - s_eye.x, target.y - s_eye.y, target.z - s_eye.z
    local l = sqrt(dx * dx + dy * dy + dz * dz)
    if l < 1 then return 0 end
    return deg(math.acos(clamp((fx * dx + fy * dy + fz * dz) / l, -1, 1)))
end

local function survivor_error(s_eye, s_ang, p)
    local eye, o
    pcall(function() eye, o = p:EyePosition(), p:get_origin() end)
    if not eye then return nil end
    local e = aim_error(s_eye, s_ang, eye)
    if o then
        e = min(e, aim_error(s_eye, s_ang, V(o.x, o.y, o.z + 36)), aim_error(s_eye, s_ang, V(o.x, o.y, o.z + 20)))
    end
    return e, dist(eye, s_eye)
end

local function clear_view(smoker_idx, s_eye, lp, now)
    if not CFG.wall then return nil end
    local c = wall_cache[smoker_idx]
    if c and now - c.t < 0.15 then return c.clear end
    local pts = {}
    pcall(function()
        local e, o = lp:EyePosition(), lp:get_origin()
        pts[#pts + 1] = e
        if o then pts[#pts + 1] = V(o.x, o.y, o.z + 36); pts[#pts + 1] = V(o.x, o.y, o.z + 14) end
    end)
    local verdict = nil
    for _, tgt in ipairs(pts) do
        local dx, dy, dz = tgt.x - s_eye.x, tgt.y - s_eye.y, tgt.z - s_eye.z
        local l = sqrt(dx * dx + dy * dy + dz * dz)
        if l > 24 then
            local steps = max(2, floor((l - 12) / 8))
            local blocked, unknown = false, false
            for i = 1, steps do
                local f = (i / (steps + 1)) * ((l - 6) / l)
                local s = solid_at(s_eye.x + dx * f, s_eye.y + dy * f, s_eye.z + dz * f)
                if s == nil then unknown = true; break end
                if s then blocked = true; break end
            end
            if unknown then verdict = nil; break end
            if not blocked then verdict = true; break end
            verdict = false
        end
    end
    wall_cache[smoker_idx] = { t = now, clear = verdict }
    return verdict
end

local function tongue_on_me(smoker, lp, now)
    local s_eye, s_ang, alive
    pcall(function() alive = smoker:is_alive(); s_eye, s_ang = smoker:EyePosition(), smoker:get_eye_angles() end)
    if not alive or not s_eye or not s_ang then return false end
    local my_err, my_d = survivor_error(s_eye, s_ang, lp)
    if not my_err or my_d > srv.range + 50 then return false end
    if my_err > AIM_TOL + deg(math.atan(24 / max(24, my_d))) then return false end
    local my_idx = lp:entindex()
    local mate = false
    pcall(function()
        for _, sv in ipairs(EntityCache.GetGroup(GROUP_SV)) do
            if sv:entindex() ~= my_idx and sv:is_alive() then
                local e, d = survivor_error(s_eye, s_ang, sv)
                if e and d <= srv.range + 50 and e + 1.5 < my_err then mate = true end
            end
        end
    end)
    if mate then return false end
    if clear_view(smoker:entindex(), s_eye, lp, now) == false then return false end
    return true, dist(s_eye, lp:EyePosition())
end

local function smoker_aiming(lp, now)
    local found = false
    pcall(function()
        for _, p in ipairs(all_infected()) do
            if p:get_zombie_class() == ZC_SMOKER and p:is_alive() and not p:is_ghost() and not p:is_dormant() then
                if tongue_on_me(p, lp, now) then found = true end
            end
        end
    end)
    return found
end

local function smoker_by_index(i)
    local found
    pcall(function()
        for _, p in ipairs(all_infected()) do
            if p:entindex() == i then found = p end
        end
    end)
    return found
end

local function do_cut(cmd, lp, smoker)
    local ang
    pcall(function()
        if smoker and smoker:is_alive() then
            local p, y = angles_to(lp:EyePosition(), smoker:get_origin())
            ang = V(clamp(p + TILT, -89, 89), y, 0)
        else
            local a = cmd:get_viewangles()
            ang = V(clamp(a.x + TILT, -89, 89), a.y, 0)
        end
    end)
    if ang then
        local old
        pcall(function() old = cmd:get_viewangles() end)
        if set_angles(cmd, ang) and old and old.y then keep_movement(cmd, old.y, ang.y) end
    end
    pcall(function()
        local b = cmd:get_buttons() or 0
        if not has_bit(b, IN_ATTACK) then cmd:set_buttons(b + IN_ATTACK) end
    end)
end

local function finish(result, now, back_delay)
    if not tongue then return end
    if result == "cut" then stats.cuts = stats.cuts + 1
    elseif result == "dragged" then stats.dragged = stats.dragged + 1
    elseif result == "missed" then
        local sm = tongue.smoker
        local m = smoker_miss[sm]
        if m and now - m.t < 6 then m.n = m.n + 1; m.t = now else m = { n = 1, t = now }; smoker_miss[sm] = m end
        if m.n >= 2 then smoker_block[sm] = now + BACKOFF end
    end
    local t = tongue
    local first = t.pressed[1]
    dbg(string.format("%s  dist %.0fu  predicted %.2f  grabbed %s  first swing %s",
        result, t.dist or 0, t.pred_grab or 0, t.grab_t and string.format("%.2f", t.grab_t) or "-",
        (first and t.grab_t) and string.format("%+.0fms", (first - t.grab_t) * 1000) or "-"))
    stats.last = result
    tongue = nil
    preswap_block_until = max(preswap_block_until, now + 1.5)
    if swapped_by_us then switch_back_at = now + (back_delay or 0.5); post_cut_release = true end
end

local function tick(cmd, lp)
    if not CFG.enabled or not cmd or not lp then tongue = nil; fire_pending = false; return end
    local alive = false
    pcall(function() alive = lp:is_alive() and lp:get_team() == 2 end)
    if not alive then tongue = nil; fire_pending = false; return end

    local now = server_now(lp)
    if not now then return end
    refresh(now)
    local is_melee = track_weapon(lp, now)

    if fire_pending then
        fire_pending = false
        if not (tongue and tongue.grab_t) then
            wall_cache = {}
            local best, best_d
            pcall(function()
                for _, p in ipairs(all_infected()) do
                    if p:get_zombie_class() == ZC_SMOKER and p:is_alive() and not p:is_dormant() then
                        local on, d = tongue_on_me(p, lp, now)
                        if on and (smoker_block[p:entindex()] or 0) > now then on = false end
                        if on and (not best_d or d < best_d) then best, best_d = p, d end
                    end
                end
            end)
            if best then
                local fire_t = best:get_prop_float(PL, "m_flSimulationTime") or now
                local base = fire_t + best_d / srv.speed
                local pred = base + correction()
                local grace = (not CFG.switch and not melee_out(lp)) and GRACE or 0
                tongue = { smoker = best:entindex(), fire_t = fire_t, dist = best_d, pred_grab = pred, base = base,
                           win_s = pred - LEAD + grace, win_e = pred + 2 * IPT, pressed = {}, override = now < override_until }
                stats.last = "tongue coming"
                dbg(string.format("tongue from %d at you  dist %.0fu  lands in %.2fs", tongue.smoker, best_d, pred - now))
            end
        end
    end

    local owner
    pcall(function() owner = player_handle(lp:get_prop_int(PL, "m_tongueOwner")) end)
    if owner and owner ~= last_owner and tongue and tongue.smoker == owner and not tongue.grab_t then
        local sm = smoker_by_index(owner)
        local grab_t = sm and sm:get_prop_float(PL, "m_flSimulationTime") or now
        local c = grab_t - tongue.base
        if c > -0.4 and c < 0.6 then
            corr[#corr + 1] = c; if #corr > 9 then table.remove(corr, 1) end
            save_calib()
        end
        tongue.grab_t = grab_t
    end
    if not owner and last_owner and tongue and tongue.grab_t then
        finish(now - tongue.grab_t < 1.5 and "cut" or "released", now)
    end
    last_owner = owner

    if not tongue then
        local aiming = CFG.switch and smoker_aiming(lp, now)
        if now < override_until or now < preswap_block_until then
            if aiming then
                if now < override_until then override_until = max(override_until, now + 0.5) end
                if now < preswap_block_until then preswap_block_until = max(preswap_block_until, now + 0.5) end
            end
            aiming = false
        end
        if aiming then aim_since = aim_since or now else aim_since = nil end
        if aiming and now - aim_since >= PRESWAP then
            preswap_seen = now
            if not is_melee and cur_cls ~= "unknown" then swap_to_melee(now) end
            switch_back_at = nil
        elseif swapped_by_us and not switch_back_at and now - preswap_seen > 2.0 then
            switch_back_at = now
        end
        if post_cut_release then
            if not is_melee then post_cut_release = false
            else
                local b = 0
                pcall(function() b = cmd:get_buttons() or 0 end)
                if has_bit(b, IN_ATTACK) then pcall(function() cmd:set_buttons(b - IN_ATTACK) end)
                else post_cut_release = false end
            end
        end
        if switch_back_at and now >= switch_back_at then
            switch_back_at = nil
            if swapped_by_us and CFG.switch_back and is_melee then swap_back(now) end
            swapped_by_us = false
        end
        return
    end
    switch_back_at = nil

    local t_ref = tongue.grab_t or tongue.fire_t
    if t_ref and now < t_ref - 1.0 then tongue = nil; return end
    if tongue.grab_t and now - tongue.grab_t > 2.0 then finish("dragged", now); return end
    if not tongue.grab_t and now > tongue.win_e + 0.5 then finish("missed", now); return end

    local smoker = smoker_by_index(tongue.smoker)
    if not tongue.grab_t then
        local alive_s, se = true, nil
        if smoker then pcall(function() alive_s = smoker:is_alive(); se = smoker:EyePosition() end) end
        if smoker and not alive_s then finish("smoker died", now, 0.1); return end
        if se and CFG.wall then
            if clear_view(tongue.smoker, se, lp, now) == false then
                tongue.lost = (tongue.lost or 0) + 1
                if tongue.lost >= 3 then finish("lost sight", now, 1.0); return end
            else
                tongue.lost = 0
            end
        end
    end

    if tongue.override then stats.last = "you have the gun"; return end

    if cur_cls ~= "unknown" and not is_melee then
        if CFG.switch then swap_to_melee(now) end
        stats.last = "gun out"
        return
    end

    if now >= tongue.win_s and now <= tongue.win_e then
        do_cut(cmd, lp, smoker)
        tongue.pressed[#tongue.pressed + 1] = now
        stats.last = "swinging"
    elseif not tongue.grab_t and now < tongue.win_s then
        if is_melee then
            pcall(function()
                local b = cmd:get_buttons() or 0
                if has_bit(b, IN_ATTACK) then cmd:set_buttons(b - IN_ATTACK) end
            end)
        end
        stats.last = string.format("swing in %.2fs", tongue.win_s - now)
    end
end

local INSTANCE = {}

local prev_cm = on_create_move
function on_create_move(cmd, lp)
    if prev_cm then pcall(prev_cm, cmd, lp) end
    if _G.echidna_autocut_instance ~= INSTANCE then return end
    local ok, err = pcall(tick, cmd, lp)
    if not ok then dbg("error: " .. tostring(err)) end
end

    events.add_listener("ability_use", function()
        if _G.echidna_autocut_instance == INSTANCE then fire_pending = true end
    end)
    events.add_listener("round_start", function()
        if _G.echidna_autocut_instance == INSTANCE then tongue = nil; fire_pending = false; last_owner = nil end
    end)
end)

local W = { value_first = nil, broken = false, last_flip = {} }
local function btn(label) local ok, r = pcall(imgui.Button, label); return ok and r == true end
local function text(s) pcall(imgui.Text, s) end
local function detect_convention()
    if W.value_first ~= nil or W.broken then return end
    for _, fn in ipairs({ imgui.SliderInt, imgui.SliderFloat }) do
        local ok, r1, r2 = pcall(fn, "##det_" .. NAME, 0, 0, 1)
        if ok and type(r1) == "number" then W.value_first = true; return end
        if ok and type(r2) == "number" then W.value_first = false; return end
    end
    W.broken = true
end
local function toggle(key, label)
    local cur = CFG[key] and true or false
    if W.broken or W.value_first == nil then
        if btn((cur and "[ON]  " or "[off] ") .. label .. "##" .. key) then return not cur end
        return cur
    end
    local ok, r1, r2 = pcall(imgui.Checkbox, label .. "##" .. key, cur)
    if not ok then W.broken = true; return cur end
    local new
    if r2 == nil then
        if W.value_first then new = r1 else new = (r1 == true) and not cur or cur end
    elseif W.value_first then new = r1 else new = r2 end
    if type(new) ~= "boolean" then W.broken = true; return cur end
    if new ~= cur then
        local f = clock()
        if W.last_flip[key] and f - W.last_flip[key] < 0.05 then W.broken = true; return cur end
        W.last_flip[key] = f
    end
    return new
end
local function box(key, label)
    local v = toggle(key, label)
    if CFG[key] ~= v then CFG[key] = v; save_cfg() end
end

local function draw_ui()
    detect_convention()
    pcall(imgui.Spacing); text("Auto Cut"); pcall(imgui.Separator)
    box("enabled", "Enabled")
    box("switch", "Switch to melee when a smoker aims at you")
    if CFG.switch then box("switch_back", "Switch back to your gun after") end
    box("wall", "Ignore smokers behind walls")
    box("debug", "Debug in console")
    pcall(imgui.Spacing); text("Status"); pcall(imgui.Separator)
    text(string.format("Cuts %d   Dragged %d   Last: %s", stats.cuts, stats.dragged, stats.last))
    text(string.format("Timing calibrated from %d grabs (%+.0f ms)", #corr, correction() * 1000))
    if btn("Reset calibration##ac_reset") then corr = {}; save_calib() end
end

local tab_ok = false
pcall(function()
    local args = { "Auto Cut" }
    local unpack_ = unpack or table.unpack
    for _ = 1, 6 do
        local ok, err = pcall(menu.add_main_tab, unpack_(args))
        if ok then tab_ok = true; return end
        local n, want = tostring(err):match("bad argument #(%d+) to '[^']*' %((%a+) expected")
        n = tonumber(n)
        if not n or n ~= #args + 1 then return end
        if want == "string" then args[n] = "Auto Cut"
        elseif want == "function" then args[n] = draw_ui
        elseif want == "number" then args[n] = 0
        elseif want == "boolean" then args[n] = true
        else return end
    end
end)
if not tab_ok then
    local prev_paint = on_paint
    function on_paint()
        if prev_paint then pcall(prev_paint) end
        local open = false
        pcall(function() open = menu.is_open() end)
        if not open then return end
        if pcall(imgui.Begin, "Auto Cut##echidna_autocut") then pcall(draw_ui) end
        pcall(imgui.End)
    end
end
